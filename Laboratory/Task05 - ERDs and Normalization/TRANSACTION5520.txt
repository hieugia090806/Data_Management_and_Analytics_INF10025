-- Create tables
CREATE TABLE PRODUCT5520 (
  ProdID       VARCHAR(10) PRIMARY KEY,
  Brand        VARCHAR(30),
  Description  VARCHAR(100),
  PricePerOne  NUMBER(8,2),
  QtyInStock   NUMBER(5) CHECK (QtyInStock >= 0)
);

CREATE TABLE ACTION5520 (
  ActionID       NUMBER(6) PRIMARY KEY,
  ActionDateTime DATE,
  ActionType     VARCHAR(10),
  ProdID         VARCHAR(10),
  ProdQty        NUMBER(5),
  ProdCost       NUMBER(8,2),
  FOREIGN KEY (ProdID) REFERENCES PRODUCT5520(ProdID)
);
